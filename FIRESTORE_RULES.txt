rules_version = '2';

// ─────────────────────────────────────────────────────────────────────────────
// КАНОНИК ДҮРМИЙН ФАЙЛ (2026-08-25-нд FIRESTORE_RULES.fixed.txt-тэй нэгтгэгдсэн)
// — хоёр нүх нөхөгдсөн (доор "ЗАСВАР 1 / 2" гэж тэмдэглэв).
//
// Үлдсэн бүх дүрэм ХЭВЭЭР. Аппын одоогийн урсгалтай тулгаж шалгасан тул
// хууль ёсны үйлдэл ямар ч газар хаагдахгүй байх ёстой — гэхдээ энэ файлыг
// Console дээр тавихаас ӨМНӨ `firestore-rules.test.mjs`-ийг заавал ажиллуулж
// баталгаажуулна уу. Тестийн хариу: 22/22 ДАВНА (2026-08-25, emulator дээр).
// ─────────────────────────────────────────────────────────────────────────────

service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() {
      return request.auth != null;
    }

    function isFamilyMember(familyId) {
      return signedIn()
        && exists(/databases/$(database)/documents/families/$(familyId)/members/$(request.auth.uid));
    }

    function memberDoc(familyId) {
      return get(/databases/$(database)/documents/families/$(familyId)/members/$(request.auth.uid));
    }

    function isAdmin(familyId) {
      return isFamilyMember(familyId)
        && memberDoc(familyId).data.role in ['parent', 'admin'];
    }

    function familyExists(familyId) {
      return exists(/databases/$(database)/documents/families/$(familyId));
    }

    function isFamilyOwner(familyId) {
      return signedIn()
        && familyExists(familyId)
        && get(/databases/$(database)/documents/families/$(familyId)).data.ownerUid == request.auth.uid;
    }

    // Урилгын код тухайн гэр бүлд ҮНЭХЭЭР хамаарах эсэхийг шалгана.
    function inviteGrants(familyId, code) {
      return code is string
        && exists(/databases/$(database)/documents/inviteCodes/$(code))
        && get(/databases/$(database)/documents/inviteCodes/$(code)).data.familyId == familyId
        && get(/databases/$(database)/documents/inviteCodes/$(code)).data.active == true;
    }

    match /users/{uid} {
      allow read: if signedIn() && request.auth.uid == uid;
      allow create, update: if signedIn() && request.auth.uid == uid;
    }

    match /inviteCodes/{code} {
      // ─── ЗАСВАР 1: жагсаалт хаах ────────────────────────────────────────
      // Өмнө нь `allow read: if signedIn();` байсан. `read` = `get` + `list`,
      // тиймээс нэвтэрсэн хэн ч (anonymous зочин ч) inviteCodes цуглуулгыг
      // БҮТНЭЭР нь query хийж, апп дээрх БҮХ гэр бүлийн familyId, familyName,
      // ownerUid-г татаж авах боломжтой байсан.
      //
      // Кодоо мэдэж байгаа хүн нэг документ уншихад л хангалттай тул
      // `get`-ийг үлдээж, `list`-ийг хаана. Аппын код зөвхөн getDoc/
      // transaction.get хийдэг учир энэ нь юуг ч эвдэхгүй.
      allow get: if signedIn();
      allow list: if false;

      allow create: if signedIn() && request.resource.data.ownerUid == request.auth.uid;
      allow update, delete: if isAdmin(resource.data.familyId);
    }

    match /families/{familyId} {
      allow create: if signedIn() && request.resource.data.ownerUid == request.auth.uid;
      allow read: if isFamilyMember(familyId);
      // ownerUid is immutable — prevents admin from hijacking ownership via update
      allow update: if isAdmin(familyId) && request.resource.data.ownerUid == resource.data.ownerUid;
      allow delete: if signedIn() && resource.data.ownerUid == request.auth.uid;

      match /members/{uid} {
        allow read: if isFamilyMember(familyId);

        // ─── ЗАСВАР 2: гишүүнчлэлийг урилгаар хамгаалах ──────────────────
        // Өмнө нь:
        //     allow create: if signedIn() && (uid == request.auth.uid || isAdmin(familyId));
        //
        // Энэ нь нэвтэрсэн ХЭН Ч, урилга огт аваагүй байж, ДУРЫН гэр бүлд
        // өөрийгөө гишүүнээр (бүр role:'parent'-аар) бичих боломж олгож
        // байсан. Гишүүн болмогц isFamilyMember() үнэн болж, тухайн гэр
        // бүлийн бүх дата — хүүхдийн өдөр тутмын лог, mood хүртэл —
        // нээгддэг. role:'parent' бичсэн бол isAdmin() ч үнэн болж, гишүүн
        // устгах, гэр бүлийн мэдээллийг өөрчлөх эрхтэй болно.
        //
        // Одоо зөвхөн гурван зам:
        allow create: if signedIn() && (
          // 1) Эцэг эх/админ шинэ гишүүн (жишээ нь хүүхдийн профайл) нэмнэ
          isAdmin(familyId)

          // 2) Гэр бүлээ дөнгөж үүсгэж буй эзэн өөрийгөө parent болгоно.
          //    handleCreateFamily нь family болон member документыг НЭГ
          //    transaction дотор бичдэг. Firestore-ийн дүрэм transaction
          //    доторх бичилтүүдийг тухайн бичилтээс ӨМНӨХ төлөвтэй харьцуулж
          //    шалгадаг тул family документ тэр агшинд бий ч байж болно,
          //    үгүй ч байж болно — хоёуланг нь зөвшөөрөв.
          //    (Байхгүй familyId дор гишүүн үүсгэх нь юу ч нээхгүй: family
          //     документ байхгүй тул уншаад ямар ч ашиггүй.)
          || (uid == request.auth.uid
              && request.resource.data.role in ['parent', 'admin']
              && (!familyExists(familyId) || isFamilyOwner(familyId)))

          // 3) Хүчинтэй урилгын кодоор нэгдэнэ. role заавал 'member' —
          //    өөрийгөө админ болгож чадахгүй.
          || (uid == request.auth.uid
              && request.resource.data.role == 'member'
              && inviteGrants(familyId, request.resource.data.joinedWithInviteCode))
        );

        // uid is immutable — prevents document ID spoofing via admin update
        allow update: if (isAdmin(familyId) && request.resource.data.uid == resource.data.uid)
          || (signedIn()
            && uid == request.auth.uid
            && request.resource.data.diff(resource.data).changedKeys().hasOnly(['name', 'avatar', 'photoURL', 'points', 'streak', 'updatedAt', 'birthYear', 'sex', 'heightCm', 'weightKg', 'activityLevel']));
        // birthYear/sex/heightCm/weightKg/activityLevel нэмэгдэв (2026-09-12):
        // хувь хүний илчлэгийн зорилт (Mifflin-St Jeor) тооцоход хэрэглэнэ.
        // ЗӨВХӨН ӨӨРӨӨ бөглөнө — админ ч бусдын биеийн үзүүлэлтийг оруулахгүй
        // (`isAdmin` зам нь бүх талбарыг зөвшөөрдөг ч UI үүнийг санал болгохгүй).
        // Бүгд СОНГОЛТТОЙ: бөглөөгүй бол ерөнхий 2000 ккал хэвээр үлдэнэ.
        //
        // 'avatar' болон 'photoURL' нэмэгдэв (2026-09-12): гишүүн өөрийнхөө
        // emoji аватар болон профайл зургаа солих боломжтой байх ёстой.
        // Өмнө нь энэ жагсаалтад байгаагүй тул энгийн гишүүн зургаа
        // оруулахад бичилт ЧИМЭЭГҮЙ татгалзаж, зөвхөн эцэг эх/админд
        // ажилладаг байв. Хоёулаа зөвхөн харагдах байдалд нөлөөлнө —
        // эрх, оноонд огт нөлөөлөхгүй тул аюулгүйн эрсдэлгүй.
        // ЗАНАЛХИЙЛЭЛ ҮЛДСЭН: гишүүн өөрийн `points`/`streak`-ийг дурын утга
        // болгож бичиж чадна (жишээ нь 999999 оноо → шагнал шууд нээгдэнэ).
        // Энэ бол аюулгүй байдлын биш, тоглоомын шударга байдлын асуудал.
        // Жинхэнэ шийдэл нь оноог Cloud Function дээр тооцоолж, клиентээс
        // огт бичихгүй болгох. Дүрэм дээр хязгаар тавьж болох ч (жишээ нь
        // нэг удаад +200-аас ихгүй) streak-ийг өнгөрсөн өдрүүдээр дахин
        // тооцоолох үед хууль ёсны их үсрэлт гардаг тул хуурамч алдаа
        // үүсгэх эрсдэлтэй. Тиймээс энд хөндөөгүй.

        allow delete: if isAdmin(familyId);
      }

      match /tasks/{uid} {
        allow read: if isFamilyMember(familyId);
        allow create: if isAdmin(familyId) || (signedIn() && uid == request.auth.uid);
        allow update: if isAdmin(familyId) || (signedIn() && uid == request.auth.uid);
        allow delete: if isAdmin(familyId);
      }

      match /dailyLogs/{logId} {
        allow read: if isFamilyMember(familyId);
        allow create, update: if isFamilyMember(familyId)
          && (isAdmin(familyId) || request.resource.data.uid == request.auth.uid);
        allow delete: if isAdmin(familyId);
      }

      match /meals/{docId} {
        allow read: if isFamilyMember(familyId);
        allow create, update, delete: if isAdmin(familyId);
      }

      match /rewards/{rewardId} {
        allow read: if isFamilyMember(familyId);
        allow create, update, delete: if isAdmin(familyId);
      }

      match /settings/{settingId} {
        // ЗОРИУДААР ХЭВЭЭР: аппын SettingsScreen дээр мэдэгдлийн тохиргоог
        // эрхээр хаагаагүй — дурын гишүүн өөрчилж чадна. Дүрмийг isAdmin
        // болгож чангатгавал хүүхэд тохиргоо хадгалахад чимээгүй бүтэлгүйтнэ
        // (аппын алдаа console.error-оор л мэдэгдэнэ). Хэрэв зөвхөн эцэг эх
        // өөрчилдөг болгох бол UI болон дүрмийг ХАМТ өөрчлөх ёстой.
        allow read: if isFamilyMember(familyId);
        allow create, update: if isFamilyMember(familyId);
        allow delete: if isAdmin(familyId);
      }

      match /nutrition/{docId} {
        // Хоолны бүртгэл (foodLogs): settings-тэй адил ЗОРИУДААР дурын гишүүн
        // бичиж болно — гэр бүлийн хэн ч өдрийн идсэн хоолоо нэмж/устгаж
        // чадна, зөвхөн эцэг эх биш.
        allow read: if isFamilyMember(familyId);
        allow create, update: if isFamilyMember(familyId);
        allow delete: if isAdmin(familyId);
      }

      // Сар бүрийн архив (2026-09-12 нэмэгдэв). resetMonthlyProgress
      // устгахаас ӨМНӨ ЭНД snapshot бичдэг болсон (useFamilyData.js).
      // Зөвхөн admin бичнэ — гишүүн бүр өөрийн дүнг чагтлаж чадахгүй байх
      // ёстой тул members/{uid} дүрмийн адил чанга.
      match /archives/{archiveId} {
        allow read: if isFamilyMember(familyId);
        allow create, update: if isAdmin(familyId);
        allow delete: if isAdmin(familyId);
      }
    }
  }
}
